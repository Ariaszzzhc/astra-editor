cmake_minimum_required(VERSION 4.4)
project(Astra VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
# Monolithic Dawn archives are large; bound concurrent linker memory.
set_property(GLOBAL PROPERTY JOB_POOLS astra_link=2)
set(CMAKE_JOB_POOL_LINK astra_link)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
endif()
if(APPLE)
    set(CMAKE_BUILD_RPATH "@executable_path")
    set(CMAKE_INSTALL_RPATH "@executable_path")
elseif(UNIX)
    set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
    set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
option(BUILD_TESTING "Build Astra behavior tests" OFF)
option(ASTRA_BUILD_GUI_TESTS "Build native desktop/GPU acceptance" OFF)
option(ASTRA_BUILD_ENVIRONMENT_TESTS "Build installed IME/isolated clipboard acceptance" OFF)
option(ASTRA_BUILD_BENCHMARKS "Build correctness-checked performance workloads" OFF)
option(ASTRA_OFFLINE "Require cached grammar and runtime archives" OFF)

add_library(astra-options INTERFACE)
target_compile_features(astra-options INTERFACE cxx_std_23)
if(MSVC)
    target_compile_definitions(astra-options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(astra-options INTERFACE /W4 /WX /permissive- /utf-8 /EHsc)
    target_link_options(astra-options INTERFACE /INCREMENTAL:NO)
endif()

include(cmake/Dependencies.cmake)
include(cmake/Grammars.cmake)
include(cmake/Icons.cmake)
include(cmake/Targets.cmake)
include(cmake/Install.cmake)
if(BUILD_TESTING OR ASTRA_BUILD_GUI_TESTS OR ASTRA_BUILD_ENVIRONMENT_TESTS)
    enable_testing()
    include(cmake/Tests.cmake)
endif()
